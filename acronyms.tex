% ===========================================================================
%  acronyms.tex  --  list of acronyms (loaded from main.tex)
%
%  Define one acronym per line:
%      \newacronym{<label>}{<short>}{<long>}
%  Optional plural forms:
%      \newacronym[longplural={...}, shortplural={...}]{<label>}{<short>}{<long>}
%
%  Use in the text:
%      \gls{MW}       first use -> "Milky Way (MW)",  later uses -> "MW"
%      \glspl{GC}     plural      -> "globular clusters (GCs)" / "GCs"
%      \Gls{GC}       capitalised first letter (start of a sentence)
%      \Glspl{GC}     capitalised plural
%      \glsxtrshort{MW} / \glsxtrlong{MW}   always short / always long form
%      \glscite{2MASS}{Skrutskie2006}       astronomy style on first use:
%                     "Two Micron All-Sky Survey (2MASS; Skrutskie et al. 2006)"
%                     and just "2MASS" afterwards (defined in preamble.tex)
%
%  Notes
%   * ONLY the acronyms actually used in the text appear in the list, so you
%     can keep entries here that you do not use (yet).
%   * The list is sorted alphabetically by the short form automatically.
%   * In captions, section titles and the abstract/resumen use \glsxtrshort
%     (or plain text): \gls there would "use up" the first expansion.
%   * The first-use counter is reset at the start of Chapter 1 (main.tex).
%     To expand every acronym again at the start of EACH chapter, put
%     \glsresetall right after each \chapter{...}.
%   * Labels are case sensitive and must not contain special characters.
% ===========================================================================
